#include <stdio.h>
#include<math.h>
#include<stdlib.h>
int x[20],n,j;

int place(int k,int i)
{
    int j,flag;
    flag=1;
    for(j=1;j<=k-1;j++)
        if((i==x[j])||(abs(k-j)==abs(i-x[j])))
        {   
            flag=0;
            break;
        }
    return flag;
}
void nqueen(int k)
{
    int i;
    for(i=1;i<=n;i++)
        if(place(k,i)==1)
        {
            x[k]=i;
            if(k==n)
            {   
                for(j=1;j<=n;j++)
                    printf("%d",x[j]);
                printf("\n");
            } 
            else
                nqueen(k+1);
        }
}
void main()
{
    printf("Enter no. of queen");
    scanf("%d",&n);
    nqueen(1);
}
